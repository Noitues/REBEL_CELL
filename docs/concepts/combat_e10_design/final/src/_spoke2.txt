# ---- R3 spoke continuity (run-width metric: contiguous INK run through the spoke axis, tol 40; a break = run < 6 px)
def run_w(x, y, nx, ny, lim=10.0, tol=40):
    w = 0.0
    for sgn in (1, -1):
        d = 0.0
        while d < lim:
            p = im[int(round(y + sgn * ny * d)), int(round(x + sgn * nx * d))]
            if np.abs(p - INKv).sum() > tol:
                break
            d += 0.5
        w += d
    return w
